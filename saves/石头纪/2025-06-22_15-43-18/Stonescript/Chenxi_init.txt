// 角色信息
func add_player_info()
  var player_info
  var pd
  player_info = player.name
  >o@1-string.Size(player_info)/2@,-2,
  ^@player_info@
  player_info = hp+"/"+maxhp
  ^+"["+armor+"."+armor.f+"]"
  >o@1-string.Size(player_info)/2@,4,
  ^@player_info@
  ?buffs.count > 0
    pd = string.Split(buffs.string,":")
    player_info = pd[2]+pd[0]+":"+pd[3]
    >o@1-string.Size(player_info)/2@,@5@,
    ^@player_info@
  ?debuffs.count > 0
    pd = string.Split(debuffs.string,":")
    player_info = pd[2]+pd[0]+":"+pd[3]
    >o@1-string.Size(player_info)/2@,@6@,
    ^@player_info@
add_player_info()
