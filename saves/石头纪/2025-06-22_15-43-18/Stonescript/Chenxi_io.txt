// info_list=[[string,id,sec,location]]
var info_list = []
// remove_list=[index]
var remove_list = []

// 最简版本
func p(str)
  info_add(str,"_random_str_"+str,"`")

// 打印普通信息
func print(str,id)
  info_add(str,id,"`")

// 打印底部信息
func print_b(str,id)
  info_add(str,id,"b")

// 打印中央信息
func print_c(str,id)
  info_add(str,id,"c")

// 打印跟随人物的信息
func print_o(str,id)
  info_add(str,id,"o")

// 打印报错信息
func print_e(str,id)
  info_add(str,id,"e")

// 添加显示文本
func info_add(str,id,location)
  // 检查id和location
  ?type(id) ! string
  ^|type(location) ! string
    return false
  _info_add_(str,id,location)

// 每帧更新
func update()
  // 每1秒检查一次
  ?totalTime % 30 = 0
    _info_timing_()
    _info_remove_()
  // 每帧打印
  _info_print_()

// 内部函数 添加文本信息
func _info_add_(str,id,location)
  // 定义消失时长，<=1秒会闪烁
  var sec = 2
  // 输入是数组，就转字符串
  ?type(str) = array
    str = list2str(str)
  // 检查是否已经存在
  ?this.info_list.Count() > 0
    for i = 0..this.info_list.Count()-1
      ?string.Equals(
      ^this.info_list[i][1],id)
      ^&string.Equals(
      ^this.info_list[i][3],location)
        // 存在就更新信息和时间
        this.info_list[i][0] = str
        this.info_list[i][2] = sec
        return false
  // 不存在就添加到列表中
  this.info_list.Add([str,id,sec,location])
  return true

// 内部函数 减少时长并检查是否结束
func _info_timing_()
  ?this.info_list.Count() <= 0
    return false
  for i = this.info_list.Count()-1..0
    this.info_list[i][2] = 
    ^this.info_list[i][2] - 1
    ?this.info_list[i][2] <= 0
      this.remove_list.Add(i)
  return true

// 内部函数 直接删除已有显示文本
func _remove_(id_list)
  // 传字符串，包装成列表
  ?type(id_list) = string
    ?string.Equals(id_list,"")
      id_list=[]
    :
      id_list=[id_list]
  // 判断是否是列表
  :?type(id_list) ! array
    return false
  // 如果是空列表，清空所有消息
  ?id_list.Count() = 0
    this.info_list.Clear()
    return true
  // 有内容，挨个判断
  for id : id_list
    // 检查是否合法
    ?type(id) ! string
    ^|this.info_list.Count() <= 0
      return false
    // 检查是否在消息列表中，在就直接删除
    for i = this.info_list.Count()-1..0
      ?string.Equals(
      ^this.info_list[i][1],id)
        this.info_list.RemoveAt(i)
        break
  return true

// 内部函数 删除信息
func _info_remove_()
  // 检查移除列表是否为空
  ?this.remove_list.Count() <= 0
    return false
  // 排序
  this.remove_list.Sort()
  // 从后往前删
  for i = this.remove_list.Count()-1..0
    //?this.remove_list[i]
    //^< this.info_list.Count()
    this.info_list.RemoveAt(
    ^this.remove_list[i])
  this.remove_list.Clear()
  return true

// 内部函数 打印
func _info_print_()
  var num = 0
  for info : this.info_list
    num = _info_print_line_(info,num)

// 内部函数 打印每行
func _info_print_line_(info,num)
  var line
  // 中央的信息
  ?info[3] = "c"
    line="["+info[1]+"] "+info[0]
    >c
    ^@1-string.Size(line)/2@,
    ^@-num%100/10@,
    ^@line@
    return num+10
  // 跟随人物的信息
  :?info[3] = "o"
    line=info[0]
    >o
    ^@1-string.Size(line)/2@,
    ^@2-num%1000/100@,
    ^@line@
    return num+100
  // 中央底部的信息
  :?info[3] = "b"
    line="["+info[1]+"] "+info[0]
    >c
    ^@1-string.Size(line)/2@,
    ^@screen.h/2-1-num%10000/1000@,
    ^@line@
    return num+1000
  // 中央底部的信息，用于报错
  :?info[3] = "e"
    line="[Error:"+info[1]+"] "+info[0]
    >c
    ^@1-string.Size(line)/2@,
    ^@screen.h/2-1-num%10000/1000@,
    ^#red,
    ^@line@
    return num+1000
  // 中央底部的信息，用于debug
  :?info[3] = "d"
    line="[Debug:"+info[1]+"] "+info[0]
    >c
    ^@1-string.Size(line)/2@,
    ^@screen.h/2-1-num%10000/1000@,
    ^#yellow,
    ^@line@
    return num+1000
  // 左上角第二行开始的普通信息
  :
    ?info[1] = "_random_str_"
      line=info[0]
    :
      line="["+info[1]+"] "+info[0]
    >`0,@2+num%10@,
    ^@line@
    return num+1

// 内部函数 从双层列表中按序数获取信息
func _get_list_by_index_(double_list,index)
  var result = []
  for list : double_list
    result.Add(list[index])
  return result

// 内部函数 debug用
func _debug_()
  _info_add_(_get_list_by_index_(this.info_list,1),"info","d")

// id是否存在
func is_exist(id)
  ?type(id)!string
    return false
  ?this.info_list.Count() <= 0
    return false
  for info : this.info_list
    ?string.Equals(info[1],id)
      return true
  return false

// 列表转字符串
func list2str(list)
  ?type(list)!array
    return false
  var str = ""
  str = str + "["
  ?list.Count() >0
    for i = 0..list.Count()-1
      ?type(list[i]) = array
        str = str + list2str(list[i])
      :
        str = str + list[i]
      ?i ! list.Count()-1
        str = str + ","
  str = str + "]"
  return str
